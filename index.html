```javascript // Get elements from the DOM const moodSelect = document.getElementById('mood'); const showSongsButton = document.getElementById('show-songs-button'); const resultsDiv = document.getElementById('results'); // Create a template element for song rendering const songTemplate = document.createElement('template'); songTemplate.innerHTML = `

{{title}}

{{artist}}

`; // Function to render songs function renderSongs(data) { if (!data || !data.songs) return; // Clear existing content to avoid duplicates resultsDiv.innerHTML = ''; // Iterate through songs and clone template for each data.songs.forEach(song => { const clone = songTemplate.content.cloneNode(true); clone.querySelector('h2').textContent = song.title; clone.querySelector('p').textContent = song.artist; resultsDiv.appendChild(clone); }); } // Function to fetch songs async function fetchSongs(mood) { try { const response = await fetch(`api/moods/${mood}`); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } return await response.json(); } catch (error) { console.error('Error:', error); return null; } } // Function to handle mood change async function handleMoodChange() { try { const selectedMood = moodSelect.value; const data = await fetchSongs(selectedMood); renderSongs(data); } catch (error) { console.error('Error:', error); } } // Add event listeners for button click and mood change showSongsButton.addEventListener('click', handleMoodChange); moodSelect.addEventListener('change', handleMoodChange); ```
